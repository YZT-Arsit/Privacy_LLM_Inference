% macros.tex --- shared math macros and drafting aids for the AAAI paper.
% Reuses the notation conventions from ../paper_draft/latex/macros.tex and
% ../paper_draft/latex/sections/a_notation.tex, adapted for the official AAAI
% 2027 template. Every symbol here is defined with dimensions and ownership in
% notation.md; keep the two files in sync.
%
% AAAI-COMPLIANCE NOTE: hyperref, cleveref-with-hyperref, geometry, and titlesec
% are forbidden by aaai2027.sty. We therefore use plain \ref/\eqref and avoid any
% link-embedding package. Do NOT add \usepackage{hyperref} or \usepackage{cleveref}.

% ---------------------------------------------------------------------------
% Plain tensors. CANONICAL threat model = PRIVATE-base: base weights W are
% proprietary/private (only the masked \Wt crosses to the untrusted GPU);
% activations are private. See notation.md for the per-symbol ownership table.
% ---------------------------------------------------------------------------
\newcommand{\X}{\mathbf{X}}          % plaintext hidden state / input activation
\newcommand{\Y}{\mathbf{Y}}          % plaintext output activation
\newcommand{\W}{\mathbf{W}}          % base weight matrix (private/proprietary; only masked \Wt crosses)
\newcommand{\Z}{\mathbf{Z}}          % generic intermediate
\newcommand{\Amat}{\mathbf{A}}       % LoRA down-projection (r x d_in)
\newcommand{\Bmat}{\mathbf{B}}       % LoRA up-projection (d_out x r)
\newcommand{\Qmat}{\mathbf{Q}}       % attention queries
\providecommand{\K}{}\renewcommand{\K}{\mathbf{K}}  % attention keys (guarded: \K may be undefined in some classes)
\newcommand{\Vmat}{\mathbf{V}}       % attention values
\newcommand{\Gmat}{\mathbf{G}}       % gradient
\newcommand{\bias}{\mathbf{b}}       % bias (public, where present)

% ---------------------------------------------------------------------------
% Mask / permutation / pad family. These objects are NOT interchangeable;
% notation.md records which is orthogonal, permutation, signed-permutation,
% invertible-dense, additive one-time, or cancellation padding.
% ---------------------------------------------------------------------------
\newcommand{\Mask}{\mathbf{N}}       % generic right-action invertible mask
\newcommand{\Nin}{\mathbf{N}_{\mathrm{in}}}
\newcommand{\Nout}{\mathbf{N}_{\mathrm{out}}}
\newcommand{\NQ}{\mathbf{N}_{Q}}     % head-space query mask
\newcommand{\NK}{\mathbf{N}_{K}}     % head-space key mask
\newcommand{\NV}{\mathbf{N}_{V}}     % value / output mask
\newcommand{\Perm}{\mathbf{P}}       % permutation matrix
\newcommand{\Umix}{\mathbf{U}}       % rank-space LoRA mixer (invertible, r_pad x r_pad)
\newcommand{\Rmask}{\mathbf{R}}      % residual-space mask

% Trusted-side additive pads and compensations.
\newcommand{\Tpad}{\mathbf{T}}       % additive one-time pad (trusted)
\newcommand{\Ct}{\mathbf{C}_{T}}     % trusted compensation term
\newcommand{\Cone}{\mathbf{C}_{1}}

% ---------------------------------------------------------------------------
% Entities, sets, and the accelerator transcript (Problem/Threat section).
% Parties are calligraphic; partition sets are blackboard-bold; tensors stay
% bold. See notation.md for the collision note (calligraphic vs bold vs
% blackboard letters that share a base glyph, e.g. P / P / P).
% ---------------------------------------------------------------------------
\newcommand{\prov}{\mathcal{P}}      % model provider (owns the proprietary checkpoint)
\newcommand{\usr}{\mathcal{U}}       % user (owns the prompt / private context)
\newcommand{\loraown}{\mathcal{L}}   % LoRA owner (adapter + private fine-tuning data)
\newcommand{\ctrl}{\mathcal{C}}      % trusted controller (trusted domain)
\newcommand{\acc}{\mathcal{G}}       % untrusted accelerator
\newcommand{\adv}{\mathcal{A}}       % adversary
\newcommand{\params}{\boldsymbol{\theta}} % proprietary parameter set
\newcommand{\rmsgain}{\boldsymbol{\gamma}} % RMSNorm / LayerNorm gains (part of \params)
\newcommand{\maskset}{\Omega}        % mask family drawn per call/session
\newcommand{\Ssec}{\mathbb{S}}       % secret-resident tensors (never leave \ctrl)
\newcommand{\Stra}{\mathbb{T}}       % transient plaintext tensors (erased after a pass)
\newcommand{\Sobs}{\mathbb{O}}       % observable (boundary-crossing) tensors
\newcommand{\Spub}{\mathbb{P}}       % public information
\newcommand{\transcript}{\tau}       % accelerator transcript (the adversary's view)
\newcommand{\arch}{\Phi}             % public architecture tuple
